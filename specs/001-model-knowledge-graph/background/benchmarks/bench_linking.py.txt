"""Benchmark: building a results-to-model index from EnergyPlus SQL output (stdlib sqlite3 + idfkit).

Usage: uv run python bench_linking.py <run dir with in.idf + eplusout.sql> [...]
"""

from __future__ import annotations

import json
import os
import sqlite3
import sys
import time

from idfkit import load_idf

AGG = {"TOTAL", "TOTAL FACILITY", "CONDITIONED TOTAL", "UNCONDITIONED TOTAL", "NOT PART OF TOTAL", "NONE"}


def build_index(doc, sql_path: str, *, with_indexes: bool) -> dict[str, int]:
    # name -> list of types, built once from idfkit collections (O(objects))
    by_name: dict[str, list[str]] = {}
    for t, coll in doc._collections.items():
        for obj in coll:
            n = obj.name
            if n:
                by_name.setdefault(n.upper(), []).append(t)
    con = sqlite3.connect(f"file:{sql_path}?mode=ro", uri=True)
    con.execute("PRAGMA query_only=1")
    stats = {"sizing": 0, "sizing_ok": 0, "keys": 0, "keys_ok": 0, "rows": 0, "rows_ok": 0, "agg": 0, "errors": 0}
    facts: dict[tuple[str, str], list[tuple]] = {}
    # 1. ComponentSizes: typed
    try:
        for ctype, cname, desc, val, units in con.execute(
            "SELECT CompType, CompName, Description, Value, Units FROM ComponentSizes"
        ):
            stats["sizing"] += 1
            key = (ctype.upper(), cname.upper())
            if ctype.upper() in {t.upper() for t in by_name.get(cname.upper(), ())}:
                stats["sizing_ok"] += 1
            facts.setdefault(key, []).append(("sizing", desc, val, units))
    except sqlite3.OperationalError:
        pass
    # 2. ReportDataDictionary keys (no timeseries data read)
    for (key,) in con.execute("SELECT DISTINCT KeyValue FROM ReportDataDictionary"):
        stats["keys"] += 1
        if key and key.upper() in by_name:
            stats["keys_ok"] += 1
    # 3. Tabular row labels: one query, pulling only distinct (report, table, row) labels
    q = (
        "SELECT DISTINCT rn.Value FROM TabularData td "
        "JOIN Strings rn ON rn.StringIndex = td.RowNameIndex"
    )
    for (row,) in con.execute(q):
        if not row:
            continue
        stats["rows"] += 1
        u = row.upper()
        if u in AGG:
            stats["agg"] += 1
        elif u in by_name:
            stats["rows_ok"] += 1
    # 4. Errors table
    try:
        for (msg, cnt) in con.execute("SELECT ErrorMessage, Count FROM Errors"):
            stats["errors"] += 1
    except sqlite3.OperationalError:
        pass
    con.close()
    stats["facts_keys"] = len(facts)
    return stats


def main() -> None:
    for run in sys.argv[1:]:
        idf = os.path.join(run, "in.idf")
        sql = os.path.join(run, "eplusout.sql")
        t0 = time.perf_counter()
        doc = load_idf(idf)
        t_load = time.perf_counter() - t0
        best = float("inf")
        for _ in range(5):
            t0 = time.perf_counter()
            stats = build_index(doc, sql, with_indexes=False)
            best = min(best, time.perf_counter() - t0)
        con = sqlite3.connect(sql)
        n_td = con.execute("SELECT COUNT(*) FROM TabularData").fetchone()[0]
        n_rdd = con.execute("SELECT COUNT(*) FROM ReportData").fetchone()[0]
        con.close()
        print(json.dumps({"run": os.path.basename(os.path.dirname(run)) or run, "sql_mb": round(os.path.getsize(sql) / 1e6, 1),
                          "tabular_cells": n_td, "timeseries_values": n_rdd, "load_idf_s": round(t_load, 3),
                          "index_build_ms": round(best * 1e3, 1), **stats}))


if __name__ == "__main__":
    main()
